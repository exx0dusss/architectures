# Internationalization (i18n)

next-intl with Zod schema factory for localized validation messages.

## Directory structure

```
src/i18n/
├── config.ts       # Locales, default locale
├── routing.ts      # next-intl routing config
├── request.ts      # getRequestConfig — loads message files
└── zod.ts          # Schema factory for i18n validation
```

## `config.ts`

```typescript
export const locales = ["uk", "ru", "en"] as const;
export type Locale = (typeof locales)[number];
export const defaultLocale: Locale = "uk";
```

## `routing.ts`

```typescript
import { defineRouting } from "next-intl/routing";
import { locales, defaultLocale } from "./config";

export const routing = defineRouting({
  locales,
  defaultLocale,
  localePrefix: "never", // Locale managed via cookies, not URL prefix
});
```

## `request.ts`

```typescript
import { getRequestConfig } from "next-intl/server";
import { routing } from "./routing";

export default getRequestConfig(async ({ requestLocale }) => {
  let locale = await requestLocale;
  if (!locale || !routing.locales.includes(locale as any)) {
    locale = routing.defaultLocale;
  }

  return {
    locale,
    messages: (await import(`../../messages/${locale}.json`)).default,
  };
});
```

## `zod.ts` — Schema factory for localized validation

This is the key building block. It allows Zod schemas to use translated error messages.

```typescript
import { z } from "zod";

// Translation function signature
export type TranslateFn = (key: string, values?: Record<string, unknown>) => string;

// No-op translator for server-side use (returns key as-is)
export const identityTranslate: TranslateFn = (key) => key;

// Factory that creates Zod helpers with localized messages
export function createSchemaFactory(t: TranslateFn) {
  return {
    requiredString: () =>
      z
        .string()
        .trim()
        .min(1, { message: t("validation.required") }),

    requiredStringNoTrim: () =>
      z.string().min(1, { message: t("validation.required") }),

    requiredStringLength: (min: number, max: number) =>
      z
        .string()
        .trim()
        .min(min, { message: t("validation.minLength", { min }) })
        .max(max, { message: t("validation.maxLength", { max }) }),

    authEmail: () =>
      z
        .string()
        .trim()
        .min(1, { message: t("validation.required") })
        .email({ message: t("validation.invalidEmail") }),

    authPassword: () =>
      z
        .string()
        .min(8, { message: t("validation.minLength", { min: 8 }) })
        .max(128, { message: t("validation.maxLength", { max: 128 }) }),

    minLength: (min: number) => ({
      message: t("validation.minLength", { min }),
    }),

    maxLength: (max: number) => ({
      message: t("validation.maxLength", { max }),
    }),

    invalidEmail: () => ({
      message: t("validation.invalidEmail"),
    }),

    invalidUrl: () => ({
      message: t("validation.invalidUrl"),
    }),

    minNumber: (min: number) => ({
      message: t("validation.minNumber", { min }),
    }),

    maxNumber: (max: number) => ({
      message: t("validation.maxNumber", { max }),
    }),

    positiveNumber: () => ({
      message: t("validation.positiveNumber"),
    }),
  };
}
```

## Usage in schemas

```typescript
// In schema.ts — factory pattern
export function createCreateFlowSchema(t: TranslateFn) {
  const f = createSchemaFactory(t);
  return z.object({
    name: f.requiredString().max(255, f.maxLength(255)),
    botId: f.requiredStringNoTrim(),
    email: f.authEmail(),
  });
}

// Default instance for server actions (no translation needed)
export const createFlowSchema = createCreateFlowSchema(identityTranslate);
export type CreateFlow = z.infer<typeof createFlowSchema>;
```

## Usage in components

```typescript
"use client";
import { useTranslations } from "next-intl";
import { createCreateFlowSchema } from "~/services/main/flows/schema";

function CreateFlowForm() {
  const t = useTranslations("validation");

  // Schema with localized error messages
  const schema = createCreateFlowSchema(t);

  const form = useAppForm({
    defaultValues: { name: "", botId: "" },
    validators: { onChange: schema },
    // ...
  });
}
```

## Server vs client usage

```typescript
// Server components
import { getTranslations } from "next-intl/server";
const t = await getTranslations("flows");
return <h1>{t("title")}</h1>;

// Client components
import { useTranslations } from "next-intl";
const t = useTranslations("flows");
return <h1>{t("title")}</h1>;
```
